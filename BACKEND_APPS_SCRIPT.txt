/* ═══ PMT 3.21.0 — podpis, wersja, limity ════════════════════════ */

// Kilka sekretów naraz: NOWY (z sekret.txt w paczce — OBRÓT SEKRETU na
// górze tej instrukcji) i STARY (wpisany w kodzie 3.21.0–3.23.0; jeśli
// wisi tu jeszcze ten z 3.20.x — też jest stary). W miejsce <...> wpisz
// prawdziwe wartości — w tym pliku ich celowo nie ma.
// Stare USUŃ z tej listy, gdy cały zespół zaktualizuje program.
var SEKRETY_PMT = [
  '<NOWY_SEKRET>',
  '<STARY_SEKRET_Z_KODU_3.21.0-3.23.0>'
];

// Akcje wysyłane WYŁĄCZNIE przez program na komputerze — tylko one mogą
// wymagać podpisu. Aplikacja na telefonie ich nie używa (sprawdzone:
// pmt_wizyty.html wysyła analiza_polki, logowanie, moje_sklepy,
// ostatnia_wizyta, produkty, zapisz_planogram, zapisz_produkt,
// zapisz_wizyte, zgloszenie, zmien_haslo — i żadnej z poniższych).
// reset_hasla jest tu najważniejsze: po tej zmianie hasła nie da się
// wyzerować z przeglądarki, tylko z podpisanego programu.
var AKCJE_PODPISANE = ['puls', 'sesja', 'reset_hasla'];

// Odcięcie starych wersji programu. Puste = nie odcinamy nikogo.
// Wpisz '3.21.0' dopiero wtedy, gdy zespół ma już nową wersję
// (zalecane: 14 dni po publikacji wydania).
var WYMAGANA_WERSJA = '';

function _pmtHex(sig) {
  return sig.map(function (b) {
    return ('0' + (b & 255).toString(16)).slice(-2);
  }).join('');
}

function weryfikujPodpis(d) {
  try {
    var czas = Number(d.klucz_czas || 0);
    if (!czas || Math.abs(Date.now() / 1000 - czas) > 600) return false;
    var baza = String(d.kod || '') + '|' + String(d.akcja || '') + '|' + czas;
    var podany = String(d.podpis || '');
    for (var i = 0; i < SEKRETY_PMT.length; i++) {
      var sig = Utilities.computeHmacSha256Signature(baza, SEKRETY_PMT[i]);
      if (_pmtHex(sig) === podany) return true;
    }
    return false;
  } catch (e) { return false; }
}

function wersjaZaStara(w) {
  if (!WYMAGANA_WERSJA) return false;        // odcinanie wyłączone
  // BRAK POLA „wersja" = NIE ODCINAMY. Aplikacja na telefonie
  // (pmt_wizyty.html) nigdy go nie wysyła — zwracanie tu „true"
  // wyłączyłoby ją w całości. Stare wersje programu na komputerze
  // odcinamy inaczej: przez MINIMALNA_WERSJA w logowanie() (KROK 4),
  // gdzie warunek obejmuje wyłącznie żądania z zrodlo="program".
  if (!w) return false;
  var a = String(w).split('.'), b = WYMAGANA_WERSJA.split('.');
  for (var i = 0; i < 3; i++) {
    var x = Number(a[i] || 0), y = Number(b[i] || 0);
    if (x > y) return false;
    if (x < y) return true;
  }
  return false;
}
